#!/usr/bin/env python3
"""Isolated development DNS pilot; no tailnet DNS changes."""

import argparse
import hashlib
import ipaddress
import json
import signal
import socket
import subprocess
import time

from common import STATE, apply, guard, load, lock, run, save

NS = "shared-dev-dns-pilot"
NAME = "dns-pilot"
ZONE = "pilot.dev.test"
OWNER = {"app.kubernetes.io/managed-by": "shared-dev-dns-pilot"}
IMAGE = "docker.io/rancher/mirrored-coredns-coredns@sha256:900f9c109f7a33545d3c811516e8376df9019147b750f5ce3e254468769176ea"


def get(kind, name=None, namespace=None):
    args = ["kubectl", "get", kind]
    if name:
        args.append(name)
    if namespace:
        args += ["-n", namespace]
    elif not name:
        args.append("-A")
    args += ["--ignore-not-found", "-o", "json"]
    raw = run(args, capture=True)
    return json.loads(raw) if raw.strip() else None


def owned(obj):
    if obj and obj["metadata"].get("labels", {}).get("app.kubernetes.io/managed-by") != OWNER["app.kubernetes.io/managed-by"]:
        raise SystemExit("Refusing to change a resource not owned by the DNS pilot")


def meta(name, ns=True):
    result = {"name": name, "labels": OWNER.copy()}
    if ns:
        result["namespace"] = NS
    return result


def objects(host, node):
    core = f"""{ZONE}:53 {{
    bind {host}
    errors
    log
    acl {{
        allow net 100.64.0.0/10
        block
    }}
    file /etc/coredns/pilot.zone {ZONE} {{
        reload 5s
    }}
}}
"""
    zone = f"""$ORIGIN {ZONE}.
$TTL 5
@ IN SOA ns.{ZONE}. hostmaster.{ZONE}. (2026100101 60 30 3600 5)
@ IN NS ns.{ZONE}.
ns IN A {host}
probe IN A {host}
*.probe IN A {host}
proof IN TXT "shared-dev-dns-pilot; Mongo endpoints managed by chart-infra"
mongo IN TXT "replicaSet=rs0&authSource=admin"
member-0.mongo IN A {host}
member-1.mongo IN A {host}
member-2.mongo IN A {host}
_mongodb._tcp.mongo IN SRV 0 0 27017 member-0.mongo.{ZONE}.
_mongodb._tcp.mongo IN SRV 0 0 27017 member-1.mongo.{ZONE}.
_mongodb._tcp.mongo IN SRV 0 0 27017 member-2.mongo.{ZONE}.
"""
    checksum = hashlib.sha256((core + zone).encode()).hexdigest()
    return [
        {"apiVersion": "v1", "kind": "Namespace", "metadata": {
            **meta(NS, False), "labels": {**OWNER, "pod-security.kubernetes.io/enforce": "privileged"},
            "annotations": {"shared-dev/purpose": "Host-network DNS pilot; non-root container; no host files, API token, or privileged container"}}},
        {"apiVersion": "v1", "kind": "ConfigMap", "metadata": meta(NAME),
         "data": {"Corefile": core, "pilot.zone": zone}},
        {"apiVersion": "apps/v1", "kind": "Deployment", "metadata": meta(NAME), "spec": {
            "replicas": 1, "strategy": {"type": "Recreate"},
            "selector": {"matchLabels": {"app": NAME}},
            "template": {"metadata": {"labels": {"app": NAME, **OWNER}, "annotations": {"shared-dev/config-sha256": checksum}},
                "spec": {"nodeSelector": {"kubernetes.io/hostname": node},
                    "hostNetwork": True, "dnsPolicy": "Default", "automountServiceAccountToken": False,
                    "terminationGracePeriodSeconds": 5,
                    "securityContext": {"runAsNonRoot": True, "runAsUser": 65532, "runAsGroup": 65532,
                                        "seccompProfile": {"type": "RuntimeDefault"}},
                    "containers": [{"name": "coredns", "image": IMAGE, "imagePullPolicy": "IfNotPresent",
                        "args": ["-conf", "/etc/coredns/Corefile"],
                        "securityContext": {"allowPrivilegeEscalation": False, "readOnlyRootFilesystem": True,
                                            "capabilities": {"drop": ["ALL"], "add": ["NET_BIND_SERVICE"]}},
                        "ports": [{"name": "dns-udp", "containerPort": 53, "protocol": "UDP"},
                                  {"name": "dns-tcp", "containerPort": 53, "protocol": "TCP"}],
                        "readinessProbe": {"tcpSocket": {"host": host, "port": 53}, "periodSeconds": 2},
                        "volumeMounts": [{"name": "config", "mountPath": "/etc/coredns", "readOnly": True}]}],
                    "volumes": [{"name": "config", "configMap": {"name": NAME}}]}}}},
    ]


def snapshot():
    result = {}
    for o in get("deployments,statefulsets,daemonsets,services")["items"]:
        m = o["metadata"]
        if m.get("namespace") in (NS, "shared-dev-mongo-pilot"):
            continue
        result[f'{m.get("namespace")}/{o["kind"]}/{m["name"]}'] = {
            "uid": m["uid"], "spec_sha256": hashlib.sha256(json.dumps(o["spec"], sort_keys=True).encode()).hexdigest()}
    cm = get("configmap", "coredns", "kube-system")
    result["kube-system/ConfigMap/coredns"] = {"uid": cm["metadata"]["uid"],
        "data_sha256": hashlib.sha256(json.dumps(cm["data"], sort_keys=True).encode()).hexdigest()}
    return result


def query(host, name, kind, tcp=False, source=None):
    args = ["dig", "@" + host, name, kind, "+time=2", "+tries=1", "+noall", "+comments", "+answer"]
    if tcp:
        args.append("+tcp")
    if source:
        args += ["-b", source]
    p = subprocess.run(args, text=True, capture_output=True, timeout=6)
    return {"command": args, "exit_code": p.returncode, "stdout": p.stdout, "stderr": p.stderr}


def check(host):
    checks = []
    for tcp in (False, True):
        for name, kind, expected in [("probe." + ZONE, "A", host),
                                      ("proof." + ZONE, "TXT", "shared-dev-dns-pilot"),
                                      ("mongo." + ZONE, "TXT", "replicaSet=rs0&authSource=admin"),
                                      ("_mongodb._tcp.mongo." + ZONE, "SRV", "member-2.mongo." + ZONE)]:
            r = query(host, name, kind, tcp)
            r["passed"] = r["exit_code"] == 0 and "status: NOERROR" in r["stdout"] and expected in r["stdout"]
            if kind == "SRV":
                r["passed"] &= all(f"member-{i}.mongo.{ZONE}." in r["stdout"] for i in range(3))
            checks.append(r)
        r = query(host, "missing." + ZONE, "A", tcp)
        r["passed"] = "status: NXDOMAIN" in r["stdout"]
        checks.append(r)
        r = query(host, "example.com", "A", tcp)
        r["passed"] = r["exit_code"] == 0 and any(f"status: {code}" in r["stdout"] for code in ("REFUSED", "SERVFAIL"))
        checks.append(r)
    node = get("nodes")["items"][0]
    lan = next(a["address"] for a in node["status"]["addresses"] if a["type"] == "InternalIP")
    for tcp in (False, True):
        r = query(host, "probe." + ZONE, "A", tcp, lan)
        r["passed"] = "status: REFUSED" in r["stdout"]
        checks.append(r)
        r = query(lan, "probe." + ZONE, "A", tcp)
        r["passed"] = r["exit_code"] != 0
        checks.append(r)
    current = snapshot()
    before = load("dns-pilot/before.json")
    unchanged = current == before
    report = {"time": time.strftime("%Y-%m-%dT%H:%M:%S%z"), "checks": checks,
        "unrelated_workload_specs_and_cluster_dns_unchanged": unchanged,
        "passed": unchanged and all(c["passed"] for c in checks),
        "not_tested_by_this_command": ["remote/client checks; see macos-*-user-reported.json", "actual Compass TLS and replica-set failover"]}
    path = save("dns-pilot/verification.json", report)
    print(json.dumps({"passed": report["passed"], "checks": len(checks), "unrelated_unchanged": unchanged, "evidence": str(path)}, indent=2))
    if not report["passed"]:
        for c in checks:
            if not c["passed"]:
                print(json.dumps(c, indent=2))
        raise SystemExit(1)


def scale(count):
    owned(get("deployment", NAME, NS))
    run(["kubectl", "-n", NS, "scale", "deployment/" + NAME, "--replicas=" + str(count)])
    if count:
        run(["kubectl", "-n", NS, "rollout", "status", "deployment/" + NAME, "--timeout=45s"])
    else:
        run(["kubectl", "-n", NS, "wait", "--for=delete", "pod", "-l", "app=" + NAME, "--timeout=30s"])


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("action", choices=["render", "up", "check", "status", "down", "outage"])
    p.add_argument("--seconds", type=int, default=30)
    a = p.parse_args()
    if not 5 <= a.seconds <= 60:
        p.error("outage duration must be 5–60 seconds")
    guard()
    held = lock("dns-pilot")
    host = str(ipaddress.IPv4Address(load("host.json")["tailscale"]))
    owned(get("namespace", NS))
    for kind in ("deployment", "configmap"):
        owned(get(kind, NAME, NS))
    manifest = objects(host, load("host.json")["node"])
    if a.action == "render":
        print(json.dumps({"apiVersion": "v1", "kind": "List", "items": manifest}, indent=2))
    elif a.action == "up":
        ts = json.loads(run(["tailscale", "status", "--json"], capture=True))
        if ts.get("BackendState") != "Running" or host not in ts.get("TailscaleIPs", []):
            raise SystemExit("Registered Tailscale address is not active")
        deployment = get("deployment", NAME, NS)
        if not deployment or not deployment.get("status", {}).get("readyReplicas"):
            # Check listeners without privileges; deployment will own the low port.
            listeners = run(["ss", "-lntu", "( sport = :53 )"], capture=True)
            if host + ":53" in listeners or "0.0.0.0:53" in listeners or "*:53" in listeners:
                raise SystemExit("DNS endpoint already has a listener; inspect ownership first")
        if not (STATE / "dns-pilot/before.json").exists():
            save("dns-pilot/before.json", snapshot())
        path = save("dns-pilot/manifest.json", {"apiVersion": "v1", "kind": "List", "items": manifest})
        apply([manifest[0]], dry=True)
        apply([manifest[0]])
        apply(manifest[1:], dry=True)
        apply(manifest[1:])
        run(["kubectl", "-n", NS, "rollout", "status", "deployment/" + NAME, "--timeout=45s"])
        print("Manifest: " + str(path))
    elif a.action == "check":
        check(host)
    elif a.action == "status":
        run(["kubectl", "-n", NS, "get", "deployment,pod", "-o", "wide"])
    elif a.action == "down":
        scale(0)
    elif a.action == "outage":
        d = get("deployment", NAME, NS)
        if not d or d["spec"].get("replicas") != 1 or not d.get("status", {}).get("readyReplicas"):
            raise SystemExit("Pilot must be ready before an outage test")
        def interrupted(signum, frame):
            raise KeyboardInterrupt
        signal.signal(signal.SIGTERM, interrupted)
        try:
            scale(0)
            print(f"DNS pilot is DOWN for {a.seconds}s; only {ZONE} is affected. Check your laptop now.", flush=True)
            time.sleep(a.seconds)
        finally:
            scale(1)
            print("DNS pilot restored. If this process is killed forcibly, run dns-pilot.py up.", flush=True)


if __name__ == "__main__":
    main()
