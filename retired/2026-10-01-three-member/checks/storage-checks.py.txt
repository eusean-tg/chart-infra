#!/usr/bin/env python3
"""Exercise startup guards against disposable fake mounts, never live Mongo files."""
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import tempfile
import sys
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

spec = importlib.util.spec_from_file_location('mongo_pilot', Path(__file__).resolve().parents[1] / 'mongo-pilot.py')
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
m.guard()
inv = m.inventory()
script = next(o for o in m.manifests(inv) if o['kind'] == 'ConfigMap')['data']['start.sh']
results = {}
with tempfile.TemporaryDirectory(prefix='mongo-guard-') as td:
    root = Path(td)
    for case in ('missing-volume-marker', 'wrong-volume-marker', 'missing-member-marker', 'wrong-member-marker'):
        p = root / case
        for d in ('volume/default/member-0', 'db', 'config'):
            (p / d).mkdir(parents=True)
        (p / 'config/start.sh').write_text(script)
        (p / 'config/marker-0').write_text(inv['markers'][0])
        if case != 'missing-volume-marker':
            (p / 'volume/.shared-dev-volume').write_text('wrong' if case == 'wrong-volume-marker' else inv['volume_marker'])
        if case != 'missing-member-marker':
            (p / 'db/.shared-dev-member').write_text('wrong' if case == 'wrong-member-marker' else inv['markers'][0])
        (p / 'volume/default/member-0/.shared-dev-member').write_text(inv['markers'][0])
        args = ['docker', 'run', '--rm', '--network', 'none', '--user', '1000:1000', '--read-only', '--cap-drop', 'ALL',
                '--security-opt', 'no-new-privileges', '--env', 'POD_INDEX=0', '--env', 'VOLUME_MARKER=' + inv['volume_marker']]
        for src, dst in [('volume', '/volume'), ('db', '/data/db'), ('config', '/config')]:
            args += ['--mount', f'type=bind,src={p/src},dst={dst},readonly']
        args += ['--entrypoint', '/bin/sh', m.IMAGE, '/config/start.sh']
        result = subprocess.run(args, text=True, capture_output=True, timeout=15)
        assert result.returncode == 1, (case, result.returncode, result.stderr)
        assert not (p / 'db/WiredTiger').exists()
        results[case] = 'rejected before mongod'
original = m.run
for fake in ('/ ext4-wrong-device', '/mnt/hdd wrong-uuid'):
    m.run = lambda *args, **kwargs: fake
    try:
        m.hdd_guard()
    except SystemExit:
        results[fake] = 'helper rejected'
    else:
        raise AssertionError('HDD guard accepted wrong mount')
m.run = original
m.inventory()
m.save('mongo-pilot/storage-guards.json', {'passed': True, 'checks': results, 'live_files_modified': False})
print(json.dumps(results, indent=2))
