#!/usr/bin/env python3
"""Probe all Mongo Services from a disposable namespace with unrestricted egress."""
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
spec=importlib.util.spec_from_file_location('mongo_pilot',Path(__file__).resolve().parents[1]/'mongo-pilot.py')
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
m.guard();held=m.lock('mongo-pilot')
ns='shared-dev-mongo-probe'
if m.kget('namespace',ns,None):
    raise SystemExit('Probe namespace already exists; inspect instead of replacing')
namespace={'apiVersion':'v1','kind':'Namespace','metadata':{'name':ns,'labels':m.OWNER}}
addresses=[m.kget('service',f'member-{i}')['spec']['clusterIP'] for i in range(3)]
script='set -eu\n'
for ip in addresses:
    script += f'if timeout 3 bash -c "exec 3<>/dev/tcp/{ip}/27017"; then echo UNEXPECTED-ACCESS; exit 1; else echo blocked-{ip}; fi\n'
pod={'apiVersion':'v1','kind':'Pod','metadata':{'name':'probe','namespace':ns,'labels':m.OWNER},'spec':{
    'restartPolicy':'Never','automountServiceAccountToken':False,
    'securityContext':{'runAsNonRoot':True,'runAsUser':1000,'runAsGroup':1000,'seccompProfile':{'type':'RuntimeDefault'}},
    'containers':[{'name':'probe','image':m.IMAGE,'command':['bash','-c',script],
        'securityContext':{'allowPrivilegeEscalation':False,'readOnlyRootFilesystem':True,'capabilities':{'drop':['ALL']}}}]}}
try:
    m.apply([namespace]);m.apply([pod])
    m.run(['kubectl','-n',ns,'wait','--for=jsonpath={.status.phase}=Succeeded','pod/probe','--timeout=45s'])
    logs=m.run(['kubectl','-n',ns,'logs','probe'],capture=True)
    assert all('blocked-'+ip in logs for ip in addresses)
    m.save('mongo-pilot/isolation.json',{'passed':True,'source':'disposable namespace with no egress restrictions','blocked_cluster_ips':addresses,'output':logs})
    print(logs)
finally:
    # This namespace contains only our disposable probe, never persistent data.
    o=m.kget('namespace',ns,None)
    if o:
        m.owned(o)
        m.run(['kubectl','delete','namespace',ns,'--wait=true','--timeout=45s'])
